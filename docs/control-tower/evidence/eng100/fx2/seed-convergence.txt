## Queries
-- posts
SELECT a.display_name, p.like_count, p.share_count, p.comment_count, p.save_count, (SELECT count(*) FROM social.reactions r WHERE r.post_id=p.id AND r.type='LIKE') likes_rows, (SELECT count(*) FROM social.comments c WHERE c.post_id=p.id AND c.deleted_at IS NULL) comment_rows FROM social.posts p JOIN social.social_accounts a ON a.id=p.author_id WHERE a.display_name IN ('Al-Haramain Ground Services','PT Baitussalam Tours','Kaaba Travel & Tours') ORDER BY 1,2
-- vendors
SELECT t.slug, v.name, v.status, (SELECT count(*) FROM marketplace.listings l WHERE l.vendor_id=v.id) FROM marketplace.vendors v JOIN core.tenants t ON t.id=v.tenant_id WHERE v.name IN ('Makkah Grand Hotels','Madinah Comfort Stays','Haramain Transport Co','Nusuk Visa Partners','Al-Haramain Ground Services') ORDER BY 2,1
-- accounts
SELECT display_name, post_count, follower_count, following_count, (SELECT count(*) FROM social.posts p WHERE p.author_id=a.id AND p.deleted_at IS NULL) posts_rows FROM social.social_accounts a WHERE display_name IN ('Al-Haramain Ground Services','PT Baitussalam Tours','Kaaba Travel & Tours') ORDER BY 1
